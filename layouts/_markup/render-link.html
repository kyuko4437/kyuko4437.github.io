{{- /* 本文中のリンクの描画ルール。これがこのサイトの仕掛けの中心。 リンク先のページが存在する → 普通の青リンク リンク先のページが存在しない → 赤リンク（class="new"） 外部サイトへのリンク → 青リンク + 矢印アイコン 記事側は普通に [表示名](/tansaki/) と書くだけでよい。 [表示名](/#tansaki) のような見出しへのリンクにも対応している。 */ -}} {{- $d := .Destination -}} {{- if or (hasPrefix $d "http://") (hasPrefix $d "https://") (hasPrefix $d "//") -}} {{ .Text | safeHTML }} {{- else -}} {{/* 「/tansaki/#開発」のような指定を、ページの部分と見出しの部分に分ける */}} {{- $path := $d -}} {{- $frag := "" -}} {{- if in $d "#" -}} {{- $sp := split $d "#" -}} {{- $path = index $sp 0 -}} {{- $frag = printf "#%s" (delimit (after 1 $sp) "#") -}} {{- end -}} {{- if eq $path "" -}} {{/* 同じページ内へのジャンプ（脚注など）。存在判定はしない */}} {{ .Text | safeHTML }} {{- else -}} {{- $p := site.GetPage $path -}} {{- if $p -}} {{ .Text | safeHTML }} {{- else -}} {{ .Text | safeHTML }} {{- end -}} {{- end -}} {{- end -}}
